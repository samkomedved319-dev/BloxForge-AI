   514→function UsersTab() {
   515→  const [users, setUsers] = useState<AdminUser[]>([]);
   516→  const [loading, setLoading] = useState(true);
   517→  const [query, setQuery] = useState("");
   518→  const [busyId, setBusyId] = useState<string | null>(null);
   519→
   520→  const load = useCallback(async () => {
   521→    setLoading(true);
   522→    try {
   523→      const res = await fetch("/api/admin/users", { cache: "no-store" });
   524→      const data = await jsonOrThrow(res);
   525→      setUsers((data as any).users ?? []);
   526→    } catch (e) {
   527→      toast.error((e as Error).message || "Failed to load users");
   528→    } finally {
   529→      setLoading(false);
   530→    }
   531→  }, []);
   532→
   533→  useEffect(() => {
   534→    load();
   535→  }, [load]);
   536→
   537→  const filtered = useMemo(() => {
   538→    const q = query.trim().toLowerCase();
   539→    if (!q) return users;
   540→    return users.filter(
   541→      (u) =>
   542→        u.email.toLowerCase().includes(q) ||
   543→        (u.name?.toLowerCase().includes(q) ?? false),
   544→    );
   545→  }, [users, query]);
   546→
   547→  async function patchUser(
   548→    id: string,
   549→    body: Record<string, unknown>,
   550→    msg: string,
   551→  ) {
   552→    setBusyId(id);
   553→    try {
   554→      const res = await fetch(`/api/admin/users/${id}`, {
   555→        method: "PATCH",
   556→        headers: { "Content-Type": "application/json" },
   557→        body: JSON.stringify(body),
   558→      });
   559→      await jsonOrThrow(res);
   560→      toast.success(msg);
   561→      await load();
   562→    } catch (e) {
   563→      toast.error((e as Error).message || "Update failed");
   564→      await load();
   565→    } finally {
   566→      setBusyId(null);
   567→    }
   568→  }
   569→
   570→  async function deleteUser(id: string) {
   571→    setBusyId(id);
   572→    try {
   573→      const res = await fetch(`/api/admin/users/${id}`, {
   574→        method: "DELETE",
   575→      });
   576→      await jsonOrThrow(res);
   577→      toast.success("User deleted");
   578→      await load();
   579→    } catch (e) {
   580→      toast.error((e as Error).message || "Delete failed");
   581→    } finally {
   582→      setBusyId(null);
   583→    }
   584→  }
   585→
   586→  return (
   587→    <motion.div
   588→      initial={{ opacity: 0, y: 12 }}
   589→      animate={{ opacity: 1, y: 0 }}
   590→      transition={{ duration: 0.4 }}
   591→      className="space-y-4"
   592→    >
   593→      <div className="flex flex-col items-start justify-between gap-3 sm:flex-row sm:items-center">
   594→        <div className="relative w-full sm:max-w-xs">
   595→          <Search className="pointer-events-none absolute left-3 top-1/2 size-4 -translate-y-1/2 text-muted-foreground" />
   596→          <Input
   597→            value={query}
   598→            onChange={(e) => setQuery(e.target.value)}
   599→            placeholder="Search by email or name…"
   600→            className="h-10 pl-9"
   601→          />
   602→        </div>
   603→        <Button
   604→          variant="ghost"
   605→          size="sm"
   606→          onClick={load}
   607→          disabled={loading}
   608→          className="gap-2 text-muted-foreground hover:text-foreground"
   609→        >
   610→          <RefreshCw className={cn("size-3.5", loading && "animate-spin")} />
   611→          Refresh
   612→        </Button>
   613→      </div>
   614→
   615→      <Card className="border-border/60 bg-card p-0">
   616→        {loading && users.length === 0 ? (
   617→          <div className="p-6">
   618→            <TabLoading label="Loading users…" bare />
   619→          </div>
   620→        ) : filtered.length === 0 ? (
   621→          <div className="p-10 text-center text-sm text-muted-foreground">
   622→            {users.length === 0
   623→              ? "No users yet."
   624→              : "No users match your search."}
   625→          </div>
   626→        ) : (
   627→          <Table>
   628→            <TableHeader>
   629→              <TableRow className="border-border/60 hover:bg-transparent">
   630→                <TableHead className="pl-4">User</TableHead>
   631→                <TableHead>Plan</TableHead>
